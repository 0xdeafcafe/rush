package main

// Copied into cmd/rush by gen.sh, run, and taken out again: runs Go's
// rush pr and rush watch over each tests/parity/*.case.json, gh faked
// with the case's answers, and writes what they said beside it as
// *.want.json for the Rust tests to say the same.

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/0xdeafcafe/rush/internal/prdrive"
	"github.com/0xdeafcafe/rush/internal/watch"
)

type parityAnswer struct {
	Match  []string `json:"match"`
	Stdout string   `json:"stdout"`
	Stderr string   `json:"stderr"`
	Exit   int      `json:"exit"`
}

type parityCase struct {
	Cmd     string         `json:"cmd"`
	Args    []string       `json:"args"`
	Session string         `json:"session"`
	On      []string       `json:"on"`
	GH      []parityAnswer `json:"gh"`
}

type parityWant struct {
	Stdout string     `json:"stdout"`
	Stderr string     `json:"stderr"`
	Code   int        `json:"code"`
	Asked  [][]string `json:"asked"`
}

func TestWriteRustParity(t *testing.T) {
	dir := os.Getenv("RUSH_PARITY")
	if dir == "" {
		t.Skip("RUSH_PARITY names the folder of cases")
	}
	files, _ := filepath.Glob(filepath.Join(dir, "*.case.json"))
	gh, runner := prdrive.GH, watch.Runner
	t.Cleanup(func() { prdrive.GH, watch.Runner = gh, runner })
	for _, f := range files {
		b, err := os.ReadFile(f)
		if err != nil {
			t.Fatal(err)
		}
		var c parityCase
		if err := json.Unmarshal(b, &c); err != nil {
			t.Fatal(f, err)
		}
		home := t.TempDir()
		t.Setenv("RUSH_HOME", home)
		t.Setenv("RUSH_SESSION", c.Session)
		if err := os.MkdirAll(filepath.Join(home, "plugins"), 0o700); err != nil {
			t.Fatal(err)
		}
		on, _ := json.Marshal(c.On)
		if err := os.WriteFile(filepath.Join(home, "plugins", "bundled-on.json"), on, 0o600); err != nil {
			t.Fatal(err)
		}
		asked := [][]string{}
		answer := func(args []string) ([]byte, []byte, int) {
			asked = append(asked, args)
			line := strings.Join(args, " ")
			for _, a := range c.GH {
				all := true
				for _, m := range a.Match {
					all = all && strings.Contains(line, m)
				}
				if all {
					return []byte(a.Stdout), []byte(a.Stderr), a.Exit
				}
			}
			return nil, []byte("no fake answer for: " + line), 1
		}
		// As the real GH says a failure: its stderr's first line, or how gh ended.
		prdrive.GH = func(_ context.Context, args ...string) ([]byte, error) {
			out, errb, code := answer(args)
			if code == 0 {
				return out, nil
			}
			msg := strings.TrimSpace(string(errb))
			if i := strings.IndexByte(msg, '\n'); i > 0 {
				msg = msg[:i]
			}
			if msg == "" {
				msg = fmt.Sprintf("exit status %d", code)
			}
			return out, fmt.Errorf("gh %s: %s", strings.Join(args[:min(2, len(args))], " "), msg)
		}
		watch.Runner = func(_ context.Context, args ...string) ([]byte, []byte, error) {
			out, errb, code := answer(append([]string{"api", "-i"}, args...))
			var err error
			if code != 0 {
				err = fmt.Errorf("exit status %d", code)
			}
			return out, errb, err
		}
		var stdout, stderr bytes.Buffer
		var code int
		switch c.Cmd {
		case "pr":
			code = prCmd(c.Args, &stdout, &stderr)
		case "watch":
			code = watchCmd(c.Args, &stdout, &stderr)
		}
		w, _ := json.MarshalIndent(parityWant{Stdout: stdout.String(), Stderr: stderr.String(), Code: code, Asked: asked}, "", " ")
		if err := os.WriteFile(strings.TrimSuffix(f, ".case.json")+".want.json", append(w, '\n'), 0o644); err != nil {
			t.Fatal(err)
		}
	}
}
