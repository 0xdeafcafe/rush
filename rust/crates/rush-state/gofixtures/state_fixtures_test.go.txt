package state

// Copied into internal/state by gen.sh to write the fixtures the Rust
// tests read: what Go rush writes, byte for byte.

import (
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/0xdeafcafe/photon/jsonx"
)

func TestWriteRustFixtures(t *testing.T) {
	out := os.Getenv("RUSH_FIXTURES")
	if out == "" {
		t.Skip("RUSH_FIXTURES unset")
	}
	write := func(name string, v any) {
		if err := writeJSON(filepath.Join(out, name), v); err != nil {
			t.Fatal(err)
		}
	}
	write("config_empty.json", Config{})
	write("overlay_empty.json", Overlay{})
	write("dispatch_empty.json", Dispatch{})

	off, on := false, true
	var c Config
	c.Feed = true
	c.CleanupDiskGB = -1
	c.CleanupRAMMinutes = 30
	c.HideMinimap = true
	c.Grid = []string{"claude/a", "codex/b"}
	c.GridBelow = true
	c.GridBelowKeys = []string{"codex/b"}
	c.GridShare = 1
	c.PixelPictures = true
	c.CompactKind = "ollama-prune"
	c.CompactModel = "m"
	c.PruneModel = "qwen"
	c.APIKeys = []string{"claude", "openai"}
	c.Folders = []Folder{{Name: "work", ConfigDir: "/x/.claude"}}
	c.FoldersImported = true
	c.Active = "work"
	c.Logins = []Login{{Name: "me", ID: "uuid-1", Email: "me@example.com", Org: "o", Profile: []byte(`{"emailAddress":"me@example.com","n":1.50}`), Borrowed: true}, {Name: "two", ID: "uuid-2"}}
	c.SignIns = []SignIn{{Kind: "codex", ID: "acct", Name: "me", Email: "e", Plan: "pro"}}
	c.Using = map[string]string{"codex": "acct", "b": "x", "A": "y"}
	c.OldProfiles = []Profile{{Name: "Default", Providers: []string{"claude", "pa"}, RunsIn: map[string]string{"pa": "pb"}, Mix: "mix", OnLimit: "handoff", Billing: "key", Account: "a", Model: "m", Effort: "high"}}
	c.DefaultProfile = "claude"
	c.OldFolderRules = []FolderRule{{Path: "~/w", Profile: "Default"}}
	c.ProfilesMovedIn = true
	c.HarnessesOff = []string{"pi"}
	c.CellModels = map[string][]string{"ollama@pi": {"q", "r"}}
	c.AtLimit = "next"
	c.RunsIn = map[string]string{"ollama": "pi"}
	c.Harnesses = map[string][]string{"ollama": {"codex"}}
	c.BuiltinProfiles = true
	c.SwitchOnLimit = "off"
	c.AgentOrder = []string{"claude", "codex"}
	c.StayOnAccount = true
	c.GroupBy = "status"
	c.SplitBy = "none"
	c.ListLook = "classic"
	c.Folds = map[string]bool{"Done": true, "active": false, "é": true, "Z": true}
	c.Dispatch = Dispatch{Kind: "claude", Agent: "rev", Model: "opus", Effort: "max", Permission: "plan", RunIn: "daemon", OnLimit: "auto", Lean: &off,
		Starts: map[string]Start{"codex": {Model: "gpt", Effort: "low", Mode: "full-access"}, "ollama-pi": {}}, Every: Start{Effort: "high"},
		ByHarness: map[string]Start{"pi": {Mode: "yolo"}}, HarnessStarts: true}
	c.Quiet = true
	c.DockLines = 4
	c.SideWidth = 0.35
	c.View = "split"
	c.ListOnly = true
	c.ChatFull = true
	c.EnterOn = "open"
	c.SortBy = "cost"
	c.StackAt = -1
	c.HideColumns = []string{"cpu", "ram"}
	c.Hibernate.AfterMinutes = 30
	c.Everywhere = Everywhere{Attribution: "off"}
	c.ActiveMinutes = 5
	c.JustLeftMinutes = -1
	c.JustLeftCount = 10
	c.CleanupHours = 72
	c.OwnWorktrees = map[string]bool{"/src/a": false, "/src/b": true}
	c.KeepOrphans = true
	c.RestartOnInstall = true
	c.TrimRunningTmp = true
	c.KeepTranscriptsPlain = true
	c.SubLine = "50%"
	c.HideLogo = true
	c.ColorBlind = true
	c.Theme = "dark"
	c.ShowWhitespace = true
	c.CopyOnSelect = &on
	c.SearchTranscriptsOnKey = true
	c.EnterSendsCommand = true
	c.RowKey = "space"
	c.MenuBar = true
	c.MenuBarAsked = true
	c.Onboarding = Onboarding{Steps: []string{"a"}, Plugins: []string{"p"}, Tips: []string{"t"}, Hidden: true}
	c.Advisor = true
	write("config_full.json", c)

	// Strings as Go writes them: <, > and & as they are; control
	// characters escaped.
	var s Config
	s.GroupBy = "a && b > c <d> \"q\" \\ \t\n\r\b\f\x01\x1f\x7f é 😀   "
	s.Folds = map[string]bool{"zz": true, "é": true, "￿": true, "😀": true, "a": true, "B": true}
	write("config_strings.json", s)

	// Floats as Go writes them.
	floats := map[string]float64{"a": 0.35, "b": 1, "c": 0.1 + 0.2, "d": 1e21, "e": 1e20, "f": 1e-7, "g": 0.000001, "h": -2.5, "i": 123456789.125, "j": 0, "k": 5e-324, "l": 1.7976931348623157e308, "m": 0.5, "n": 100}
	write("floats.json", floats)

	// Times as Go writes them: in their own zone, nanoseconds trimmed.
	at := time.Date(2026, 10, 28, 12, 34, 56, 123456789, time.UTC)
	plus := time.FixedZone("x", 3600+30*60)
	minus := time.FixedZone("y", -7*3600)
	o := Overlay{
		Done:   map[string]time.Time{"claude/a": at, "claude/b": at.Truncate(time.Second).In(plus)},
		Names:  map[string]string{"claude/a": "fix the build"},
		Groups: map[string]string{"claude/a": "g"},
		Moved:  map[string]string{"x": "y"},
		Seen:   map[string]time.Time{"claude/a": at.Add(-time.Millisecond * 100).In(minus), "z": time.Date(1999, 1, 2, 3, 4, 5, 100000000, time.UTC)},
		Hidden: map[string]time.Time{"claude/c": at.Add(time.Microsecond)},
	}
	write("state.json", o)

	// The cost cache: compact, as it's made.
	type file struct {
		Offset int64   `json:"offset"`
		Cost   float64 `json:"cost,omitzero"`
		Cwd    string  `json:"cwd,omitempty"`
	}
	cc := &CostCache[file]{Version: costCacheVersion, Files: map[string]*file{"/b.jsonl": {Offset: 10, Cost: 0.25}, "/a.jsonl": {Offset: 3, Cwd: "/src"}}}
	f, err := os.Create(filepath.Join(out, "costs.json"))
	if err != nil {
		t.Fatal(err)
	}
	if err := jsonx.MarshalWrite(f, cc); err != nil {
		t.Fatal(err)
	}
	f.Close()

	// Old configs, loaded and saved again by Go: what migrate makes of them
	// with the test agents registered.
	for name, old := range map[string]string{
		"migrate_accounts": `{"accounts":[{"name":"work","configDir":"/x/.claude"},{"name":"old","configDir":"/x/.claude-old"}],"stayOnAccount":true,"groupBy":"account"}`,
		"migrate_profiles": `{"dispatch":{"kind":"pa"},"switchOnLimit":"agent","agentOrder":["pb","pa"]}`,
		"migrate_movein": `{"dispatch":{"kind":"claude"},"builtinProfiles":true,
		"profiles":[{"name":"work","providers":["pa","pc"],"runsIn":{"pa":"pb"},"model":"m1","effort":"high","onLimit":"wait"}],
		"defaultProfile":"work","folderRules":[{"path":"~/w","profile":"work"}]}`,
		"migrate_plain":  `{"dispatch":{"kind":"claude","effort":"low","permission":"plan"},"agentOrder":["claude","pa"],"profiles":[{"name":"Default","providers":["claude","pa"]}],"defaultProfile":"Default","folderRules":[{"path":"~/w","profile":"Default"}]}`,
		"migrate_split":  `{"dispatch":{"kind":"ps","starts":{"ps-key":{"effort":"max"},"pa":{"mode":"plan","effort":"low"}}},"runsIn":{"ps":"pb"},"defaultProfile":"ps","groupBy":"repo"}`,
		"migrate_empty":  `{}`,
		"migrate_unknown": `{"groupBy":"folder","someFutureKey":{"x":1},"dockLines":3,"sideWidth":0.30}`,
	} {
		home := t.TempDir()
		t.Setenv("RUSH_HOME", home)
		if err := os.WriteFile(filepath.Join(out, name+".in.json"), []byte(old), 0o600); err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(filepath.Join(home, "config.json"), []byte(old), 0o600); err != nil {
			t.Fatal(err)
		}
		s := Load()
		if err := s.SaveConfig(); err != nil {
			t.Fatal(err)
		}
		b, _ := os.ReadFile(filepath.Join(home, "config.json"))
		if err := os.WriteFile(filepath.Join(out, name+".out.json"), b, 0o600); err != nil {
			t.Fatal(err)
		}
	}
}
