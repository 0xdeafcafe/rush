package health

// Copied into internal/health by gen.sh to write the fixtures the Rust
// tests read.

import (
	"os"
	"path/filepath"
	"testing"
	"time"
)

func TestWriteRustFixtures(t *testing.T) {
	out := os.Getenv("RUSH_FIXTURES")
	if out == "" {
		t.Skip("RUSH_FIXTURES unset")
	}
	t.Setenv("RUSH_HOME", t.TempDir())
	t0 := time.Date(2026, 10, 28, 12, 0, 0, 500, time.FixedZone("x", 3600))
	Answered("claude/a", t0)
	Failed("claude/a", "API Error: 401 invalid_grant sk-secret", t0.Add(2*time.Second))
	Failed("codex/b", "API Error: Connection error.", t0.Add(3*time.Second))
	Failed("codex/b", "API Error: 503", t0.Add(4*time.Second))
	Answered("pi/c", t0.Add(time.Minute))
	b, err := os.ReadFile(Path())
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(out, "health.json"), b, 0o600); err != nil {
		t.Fatal(err)
	}
}
