package settingsfile

// Copied into internal/settingsfile by gen.sh to write the fixtures the
// Rust tests read: a file as an agent wrote it, edited by Go rush.

import (
	"os"
	"path/filepath"
	"testing"
)

func TestWriteRustFixtures(t *testing.T) {
	out := os.Getenv("RUSH_FIXTURES")
	if out == "" {
		t.Skip("RUSH_FIXTURES unset")
	}
	in := "{\n\t\"model\" : \"opus\",\n  \"n\": 1.50e+2, \"neg\": -0, \"esc\": \"a\\u0026b \\/ \\u00e9 \\ud83d\\ude00\",\n\"hooks\": {\"Stop\": [ {\"command\": \"a && b > out\"} ], \"empty\": {}, \"arr\": []},\n \"env\": {\"A\": \"1\", \"DROP\": \"x\"},\n \"permissions\": {\"defaultMode\": \"plan\", \"allow\": [\"Bash(ls)\", 1, true, null]}, \"model\": \"sonnet\"\n}\n"
	if err := os.WriteFile(filepath.Join(out, "settings.in.json"), []byte(in), 0o600); err != nil {
		t.Fatal(err)
	}
	work := filepath.Join(t.TempDir(), "settings.json")
	if err := os.WriteFile(work, []byte(in), 0o600); err != nil {
		t.Fatal(err)
	}
	s, err := Load(work)
	if err != nil {
		t.Fatal(err)
	}
	_ = s.SetEnv("B", "2 <&> é")
	_ = s.SetEnv("DROP", "")
	_ = s.Set("permissions.defaultMode", "bypassPermissions")
	_ = s.Set("statusLine.type", "command")
	_ = s.Set("statusLine.command", "rush status && echo")
	_ = s.Set("new.deep.list", []any{1, 2.5, "x", map[string]any{"k": nil}})
	_ = s.Set("hooks.empty", nil)
	if err := s.Save(); err != nil {
		t.Fatal(err)
	}
	b, _ := os.ReadFile(work)
	if err := os.WriteFile(filepath.Join(out, "settings.out.json"), b, 0o600); err != nil {
		t.Fatal(err)
	}
	// An empty file, then values at the top and inside.
	empty := filepath.Join(t.TempDir(), "s.json")
	e, _ := Load(empty)
	_ = e.Set("theme", "dark")
	_ = e.Set("env.X", "1")
	if err := e.Save(); err != nil {
		t.Fatal(err)
	}
	b, _ = os.ReadFile(empty)
	if err := os.WriteFile(filepath.Join(out, "settings_new.out.json"), b, 0o600); err != nil {
		t.Fatal(err)
	}
}
